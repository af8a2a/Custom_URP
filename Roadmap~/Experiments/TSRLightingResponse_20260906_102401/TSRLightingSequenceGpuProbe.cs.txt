    public static class TSRLightingSequenceGpuProbe
    {
        [Serializable] private sealed class Step
        {
            public string scenario, variant;
            public int frame;
            public bool passed;
            public float previousState, expectedState, pendingState, accepted, acceptedAlpha, samples, resurrectionFrames;
            public Color current, previous, updated, acceptedHistory, resurrectionColor;
        }
        [Serializable] private sealed class Report
        {
            public string output, error, scope;
            public bool success;
            public int passed, failed;
            public Step[] steps;
        }
        [MenuItem("Tools/VividRP/Diagnostics/Run Temporary TSR Lighting Sequence GPU Probe")]
        public static void Run()
        {
            if (!Application.dataPath.Replace('\\', '/').Equals("E:/VividRP_Reborn/Assets", StringComparison.OrdinalIgnoreCase))
                throw new InvalidOperationException("TSR sequence probe is scoped to VividRP_Reborn.");
            var report = new Report
            {
                output = Path.Combine(Path.GetTempPath(), "tsr-lighting-sequence-gpu-" + DateTime.UtcNow.ToString("yyyyMMdd-HHmmss") + ".json"),
                scope = "Direct production Reject/Update, recursive actual updated history/resurrection RGB, count and pending alpha. Isolated 8x8 NativeAA contract checks; no scene or NUnit runner. Explicit update-only case injects an acceptance state to test pending storage with an expired cache."
            };
            var steps = new List<Step>();
            try
            {
                for (int wave = 0; wave < 2; wave++)
                {
                    if (wave != 0 && SystemInfo.graphicsDeviceType != GraphicsDeviceType.Direct3D12
                        && SystemInfo.graphicsDeviceType != GraphicsDeviceType.Vulkan) continue;
                    using var fixture = new Fixture(wave != 0);
                    RunCases(fixture, wave == 0 ? "scalar" : "wave", steps);
                }
            }
            catch (Exception exception) { report.error = exception.ToString(); }
            foreach (var step in steps) { if (step.passed) report.passed++; else report.failed++; }
            report.steps = steps.ToArray(); report.success = report.error == null && report.failed == 0;
            File.WriteAllText(report.output, JsonUtility.ToJson(report, true));
            Debug.Log("TSR lighting sequence GPU diagnostic: " + report.output + "; passed=" + report.passed + "; failed=" + report.failed);
        }

        private static void RunCases(Fixture fixture, string variant, List<Step> steps)
        {
            for (int direction = 1; direction <= 6; direction++)
                Confirm(fixture, variant, "direction-" + direction, DirectionInput(direction), direction, steps);
            var mixed = new Input { Current = new Color(0.63f, 0.85f, 0.07f), History = Gray(0.6f), ChromaPattern = true,
                NeighborhoodColorA = new Color(0.73f, 0.95f, 0.17f), NeighborhoodColorB = Gray(0.3f) };
            Confirm(fixture, variant, "combined-chroma-under-component-threshold", mixed, 3, steps);
            Confirm(fixture, variant, "recursive-stale-resurrection-bright", new Input
                { Current = Gray(1), History = Gray(0.56f), Resurrection = Gray(0.56f), ResurrectionFrames = 6,
                    LumaInstability = 1, NeighborhoodLow = 0.3f, NeighborhoodHigh = 1 }, 1, steps);
            Confirm(fixture, variant, "recursive-stale-resurrection-dark", new Input
                { Current = Gray(0.56f), History = Gray(1), Resurrection = Gray(1), ResurrectionFrames = 6,
                    LumaInstability = 1, NeighborhoodLow = 0.3f, NeighborhoodHigh = 1 }, 2, steps);

            foreach (Vector4 values in new[] { new Vector4(1, 0.56f, 0.67f, 0.695872f), new Vector4(0.56f, 1, 0.89f, 0.864128f) })
            {
                int direction = values.x > values.y ? 1 : 2;
                string suffix = direction == 1 ? "bright" : "dark";
                var direct = new Input { Current = Gray(values.x), History = Gray(values.y), PreviousState = direction * 4 + 2,
                    Resurrection = Gray(values.y), ResurrectionFrames = 6, NeighborhoodLow = 0.3f, NeighborhoodHigh = 1 };
                Snapshot soft = Record(fixture, variant, "soft-numeric-direct-" + suffix, 1, direct, 0, true, steps);
                steps[steps.Count - 1].passed &= ColorError(soft.Updated, Gray(values.z)) < 0.003f
                    && ColorError(soft.AcceptedHistory, direct.History) < 0.003f;
                Feed(direct, soft); direct.Current = direct.History;
                Snapshot recovered = Record(fixture, variant, "soft-next-ordinary-count-recovers-" + suffix, 2, direct, 0, false, steps);
                steps[steps.Count - 1].passed &= recovered.SampleCount == 5;
                var recursive = new Input { Current = Gray(values.x), History = Gray(values.y), Resurrection = Gray(values.y),
                    ResurrectionFrames = 6, NeighborhoodLow = 0.3f, NeighborhoodHigh = 1 };
                for (int frame = 1; frame <= 3; frame++)
                {
                    Snapshot current = Record(fixture, variant, "soft-numeric-recursive-" + suffix, frame,
                        recursive, frame < 3 ? direction * 4 + frame : 0, frame == 3, steps);
                    if (frame == 3) steps[steps.Count - 1].passed &= ColorError(current.Updated, Gray(values.w)) < 0.003f;
                    Feed(recursive, current);
                }
            }

            var alternating = new Input { Current = Gray(0.9f), History = Gray(0.5f) };
            for (int frame = 1; frame <= 16; frame++)
            {
                alternating.Current = Gray(frame % 2 == 1 ? 0.9f : 0.1f);
                StepAndFeed(fixture, variant, "alternating-high-amplitude", frame, alternating,
                    frame % 2 == 1 ? 5 : 9, false, steps);
            }
            var interrupted = DirectionInput(1);
            StepAndFeed(fixture, variant, "interrupted-streak", 1, interrupted, 5, false, steps);
            interrupted.Current = interrupted.History;
            StepAndFeed(fixture, variant, "interrupted-streak", 2, interrupted, 0, false, steps);
            interrupted.Current = Gray(0.8f);
            StepAndFeed(fixture, variant, "interrupted-streak", 3, interrupted, 5, false, steps);
            StepAndFeed(fixture, variant, "interrupted-streak", 4, interrupted, 6, false, steps);
            StepAndFeed(fixture, variant, "interrupted-streak", 5, interrupted, 0, true, steps);
            var reversed = new Input { Current = Gray(0.8f), History = Gray(0.5f) };
            StepAndFeed(fixture, variant, "reversed-streak", 1, reversed, 5, false, steps);
            StepAndFeed(fixture, variant, "reversed-streak", 2, reversed, 6, false, steps);
            reversed.Current = Gray(0.1f);
            StepAndFeed(fixture, variant, "reversed-streak", 3, reversed, 9, false, steps);
            StepAndFeed(fixture, variant, "reversed-streak", 4, reversed, 10, false, steps);
            StepAndFeed(fixture, variant, "reversed-streak", 5, reversed, 0, true, steps);

            Record(fixture, variant, "below-entry-no-pending", 1, ThresholdInput(0.26f, 0), 0, false, steps);
            Record(fixture, variant, "above-entry-no-pending", 1, ThresholdInput(0.29f, 0), 5, false, steps);
            Record(fixture, variant, "inside-continuation-band-count1", 1, ThresholdInput(0.26f, 5), 6, false, steps);
            Record(fixture, variant, "inside-continuation-band-count2", 1, ThresholdInput(0.26f, 6), 0, true, steps);
            Record(fixture, variant, "below-continuation-clears", 1, ThresholdInput(0.13f, 5), 0, false, steps);
            Record(fixture, variant, "new-direction-cannot-use-continuation", 1, ThresholdInput(0.26f, 9), 0, false, steps);
            Record(fixture, variant, "small-difference-no-pending", 1, ThresholdInput(0.15f, 0), 0, false, steps);
            Record(fixture, variant, "small-difference-continues-count1", 1, ThresholdInput(0.15f, 5), 6, false, steps);
            Record(fixture, variant, "small-difference-confirms-count2", 1, ThresholdInput(0.15f, 6), 0, true, steps);
            var unstableStatic = ThresholdInput(0.30f, 0); unstableStatic.LumaInstability = 1;
            Record(fixture, variant, "static-instability-does-not-raise-entry-threshold", 1, unstableStatic, 5, false, steps);
            var unstableMoving = ThresholdInput(0.30f, 6); unstableMoving.LumaInstability = 1; unstableMoving.MotionPixels = 2;
            Record(fixture, variant, "moving-instability-keeps-original-threshold", 1, unstableMoving, 0, false, steps);
            foreach (int previous in new[] { 1, 7, 9, 31 })
            {
                var input = DirectionInput(1); input.PreviousState = previous;
                Record(fixture, variant, "legacy-invalid-or-other-direction-" + previous, 1, input, 5, false, steps);
            }
            var moving = DirectionInput(1); moving.PreviousState = 6; moving.MotionPixels = 2;
            Record(fixture, variant, "moving-immediately-rejects", 1, moving, 0, true, steps);
            Record(fixture, variant, "moving-match-clears-pending", 1,
                new Input { Current = Gray(0.5f), History = Gray(0.5f), PreviousState = 6, MotionPixels = 2 }, 0, false, steps);
            var geometryEdge = DirectionInput(1); geometryEdge.DepthError = 0.02f; geometryEdge.PreviousState = 6;
            for (int frame = 1; frame <= 3; frame++)
                StepAndFeed(fixture, variant, "stationary-depth-edge-clears-and-retains", frame, geometryEdge, 0, false, steps);
            var movingEdge = DirectionInput(1); movingEdge.DepthError = 0.02f; movingEdge.PreviousState = 6; movingEdge.MotionPixels = 2;
            Record(fixture, variant, "moving-depth-edge-still-rejects-color", 1, movingEdge, 0, true, steps);
            foreach (Vector2Int offset in new[] { new Vector2Int(-3, 0), new Vector2Int(3, 0),
                new Vector2Int(0, -3), new Vector2Int(0, 3), new Vector2Int(-3, -3), new Vector2Int(3, 3) })
            {
                Input nearbyGeometry = DirectionInput(1); nearbyGeometry.PreviousState = 6;
                nearbyGeometry.NearbyDepthFeatureOffset = offset;
                for (int frame = 1; frame <= 3; frame++)
                    StepAndFeed(fixture, variant, "nearby-geometry-continuous-center-" + offset.x + "-" + offset.y,
                        frame, nearbyGeometry, 0, false, steps);
            }
            for (int i = 0; i < 16; i++)
                Record(fixture, variant, "small-noise-clears-pending", i + 1, new Input
                { Current = Gray(0.5f + ((i * 7 % 17) - 8) * 0.003125f), History = Gray(0.5f), PreviousState = 6 }, 0, false, steps);

            var expired = DirectionInput(1); expired.PreviousState = 5; expired.ResurrectionFrames = 0;
            Record(fixture, variant, "expired-cache-pending-count1", 1, expired, 6, false, steps);
            expired.PreviousState = 6;
            Record(fixture, variant, "expired-cache-pending-count2", 1, expired, 0, true, steps);
            var forced = new Input { Current = Gray(0.5f), History = Gray(0.5f), ResurrectionFrames = 0,
                ForceUpdateState = true, ForcedUpdateState = 5 };
            Snapshot forcedResult = fixture.Run(forced);
            Add(variant, "update-only-pending-survives-expired-cache", 1, forced, forcedResult, 5,
                forcedResult.PendingState == 5 && forcedResult.ResurrectionFrames == 0, steps);
            var compatible = new Input { Current = Gray(1), History = Gray(0.1f), HistorySamples = 0,
                PreviousState = 6, Resurrection = Gray(0.9f), ResurrectionFrames = 6, NeighborhoodHigh = 1 };
            Snapshot restored = fixture.Run(compatible);
            Add(variant, "invalid-primary-compatible-resurrection", 1, compatible, restored, 0,
                restored.Accepted == 0 && restored.AcceptedAlpha == 0 && restored.PendingState == 0 && Mathf.Abs(restored.Updated.r - 0.965f) < 0.003f
                && restored.ResurrectionFrames == 5 && restored.SampleCount > 1, steps);
        }

        private static Input ThresholdInput(float difference, int previous)
            => new Input { Current = Gray(1), History = Gray(1 - difference), PreviousState = previous, NeighborhoodHigh = 1 };
        private static Input DirectionInput(int direction)
        {
            if (direction <= 2) return new Input { Current = Gray(direction == 1 ? 0.8f : 0.2f), History = Gray(direction == 1 ? 0.2f : 0.8f) };
            Color a = direction <= 4 ? new Color(0.9f, 0.1f, 0.1f) : new Color(0.1f, 0.9f, 0.1f);
            Color b = direction <= 4 ? new Color(0.1f, 0.1f, 0.9f) : new Color(0.9f, 0.1f, 0.9f);
            return new Input { Current = direction % 2 == 1 ? a : b, History = direction % 2 == 1 ? b : a,
                ChromaPattern = true, NeighborhoodColorA = a, NeighborhoodColorB = b };
        }
        private static void Confirm(Fixture fixture, string variant, string scenario, Input input, int direction, List<Step> steps)
        {
            for (int frame = 1; frame <= 3; frame++)
                StepAndFeed(fixture, variant, scenario, frame, input, frame < 3 ? direction * 4 + frame : 0, frame == 3, steps);
        }
        private static void Feed(Input input, Snapshot result)
        {
            input.History = result.Updated; input.HistorySamples = result.SampleCount;
            input.Resurrection = result.ResurrectionColor; input.ResurrectionFrames = result.ResurrectionFrames;
            input.PreviousState = result.PendingState; input.HistoryCheckerboard = false;
        }
        private static void StepAndFeed(Fixture fixture, string variant, string scenario, int frame, Input input, int expectedState, bool confirmed, List<Step> steps)
        { Snapshot result = Record(fixture, variant, scenario, frame, input, expectedState, confirmed, steps); Feed(input, result); }
        private static Snapshot Record(Fixture fixture, string variant, string scenario, int frame, Input input, int expectedState, bool confirmed, List<Step> steps)
        {
            Snapshot result = fixture.Run(input);
            bool hardRejected = confirmed && input.MotionPixels > 1;
            bool passed = result.PendingState == expectedState && result.Accepted == (hardRejected ? 0 : 1);
            if (confirmed)
            {
                passed &= result.ResurrectionFrames == 0 && ColorError(result.ResurrectionColor, Color.clear) < 0.003f;
                if (hardRejected) passed &= result.AcceptedAlpha == -1 && ColorError(result.Updated, input.Current) < 0.003f;
                else passed &= result.AcceptedAlpha == -2 && result.SampleCount == 4
                    && ColorError(result.Updated, input.Current) <= ColorError(result.AcceptedHistory, input.Current) * 0.75f + 0.003f;
            }
            else passed &= result.AcceptedAlpha >= 0 && result.SampleCount > 1;
            Add(variant, scenario, frame, input, result, expectedState, passed, steps); return result;
        }
        private static float ColorError(Color a, Color b)
            => Mathf.Max(Mathf.Max(Mathf.Abs(a.r - b.r), Mathf.Abs(a.g - b.g)), Mathf.Abs(a.b - b.b));
        private static void Add(string variant, string scenario, int frame, Input input, Snapshot result, int expected, bool passed, List<Step> steps)
        {
            steps.Add(new Step { variant = variant, scenario = scenario, frame = frame, passed = passed,
                previousState = input.PreviousState, expectedState = expected, pendingState = result.PendingState,
                accepted = result.Accepted, acceptedAlpha = result.AcceptedAlpha, samples = result.SampleCount,
                resurrectionFrames = result.ResurrectionFrames, current = input.Current, previous = input.History, updated = result.Updated,
                acceptedHistory = result.AcceptedHistory, resurrectionColor = result.ResurrectionColor });
        }

        private static Color Gray(float value) => new(value, value, value, 1);

        private static Input SpatialJitterInput(int phase, float instability)
            => new Input
            {
                Current = Gray(phase == 0 ? 0.9f : 0.1f), History = Gray(phase == 0 ? 0.1f : 0.9f),
                CheckerPhase = phase, HistoryCheckerboard = true, HistoryCheckerPhase = 1 - phase,
                LumaInstability = instability
            };

        private static Input LowFrequencyLightingInput(bool brightening)
            => brightening ? new Input
            {
                Current = Gray(0.9f), History = Gray(0.5f), NeighborhoodLow = 0.18f, NeighborhoodHigh = 0.9f,
                HistoryCheckerboard = true, HistoryCheckerLow = 0.1f, HistoryCheckerHigh = 0.5f
            } : new Input
            {
                Current = Gray(0.1f), History = Gray(0.18f), NeighborhoodLow = 0.1f, NeighborhoodHigh = 0.5f,
                CheckerPhase = 1, HistoryCheckerboard = true, HistoryCheckerPhase = 1,
                HistoryCheckerLow = 0.18f, HistoryCheckerHigh = 0.9f
            };

        private static float BinomialMean(float[] values)
        {
            float sum = 0;
            for (int y = -1; y <= 1; y++) for (int x = -1; x <= 1; x++)
            {
                int pixel = (4 + y) * Fixture.Size + 4 + x;
                float luma = values[pixel * 4] * 0.25f + values[pixel * 4 + 1] * 0.5f + values[pixel * 4 + 2] * 0.25f;
                sum += luma * (x == 0 ? 2 : 1) * (y == 0 ? 2 : 1);
            }
            return sum / 16;
        }

        private sealed class Input
        {
            internal Color Current = Gray(0.8f), History = Gray(0.2f), Resurrection = Color.clear;
            internal float PreviousState;
            internal bool ForceUpdateState;
            internal float ForcedUpdateState;
            internal Color NeighborhoodColorA = new Color(0.9f, 0.1f, 0.1f), NeighborhoodColorB = new Color(0.1f, 0.1f, 0.9f);
            internal Vector2Int NearbyDepthFeatureOffset;
            internal float DepthError, MotionPixels, LumaInstability, HistorySamples = 16, HistoryDepth = 0.5f, ResurrectionFrames;
            internal float NeighborhoodLow = 0.1f, NeighborhoodHigh = 0.9f;
            internal bool ChromaPattern, HistoryMatchesInput, HistoryCheckerboard;
            internal int CheckerPhase, HistoryCheckerPhase;
            internal float HistoryCheckerLow = 0.1f, HistoryCheckerHigh = 0.9f;
        }

        private sealed class Snapshot
        {
            internal float[] CurrentColors, HistoryColors, UpdatedColors, Acceptance;
            internal Color Updated, ResurrectionColor, AcceptedHistory;
            internal float Accepted, AcceptedAlpha, SampleCount, ResurrectionFrames, PendingState;
        }

        private sealed class Fixture : IDisposable
        {
            internal const int Size = 8, PixelCount = Size * Size;
            private const int Center = 4 * Size + 4;
            private readonly ComputeShader reject, update;
            private readonly Texture2D color, history, resurrection, depth, depthError, zero, instability, motion, historyMeta, resurrectionMeta;
            private readonly RenderTexture acceptedColor, rejection, updatedColor, updatedMeta, updatedResurrectionColor, updatedResurrectionMeta;

            internal Fixture(bool waveOps = false)
            {
                reject = Load("TSRRejectShading.compute", waveOps);
                update = Load("TSRUpdateHistory.compute", waveOps);
                color = CreateInput(GraphicsFormat.R32G32B32A32_SFloat);
                history = CreateInput(GraphicsFormat.R32G32B32A32_SFloat);
                resurrection = CreateInput(GraphicsFormat.R32G32B32A32_SFloat);
                depth = CreateInput(GraphicsFormat.R32_SFloat); depthError = CreateInput(GraphicsFormat.R32_SFloat); zero = CreateInput(GraphicsFormat.R32_SFloat);
                instability = CreateInput(GraphicsFormat.R32_SFloat); motion = CreateInput(GraphicsFormat.R32G32_SFloat);
                historyMeta = CreateInput(GraphicsFormat.R32G32_SFloat); resurrectionMeta = CreateInput(GraphicsFormat.R32G32_SFloat);
                acceptedColor = CreateOutput(GraphicsFormat.R32G32B32A32_SFloat);
                rejection = CreateOutput(GraphicsFormat.R32_SFloat); updatedColor = CreateOutput(GraphicsFormat.R32G32B32A32_SFloat);
                updatedMeta = CreateOutput(GraphicsFormat.R32G32_SFloat);
                updatedResurrectionColor = CreateOutput(GraphicsFormat.R32G32B32A32_SFloat);
                updatedResurrectionMeta = CreateOutput(GraphicsFormat.R32G32_SFloat);
                SetConstant(depth, 1, 0.5f); SetConstant(zero, 1, 0);
            }

            private static ComputeShader Load(string name, bool waveOps)
            {
                var source = AssetDatabase.LoadAssetAtPath<ComputeShader>(
                    "Packages/com.vivid.render-pipelines/Shaders/Core/Private/TSR/" + name);
                if (source == null) throw new InvalidOperationException("Missing TSR compute: " + name);
                var shader = Object.Instantiate(source);
                if (waveOps) shader.EnableKeyword("VIVID_TSR_WAVE_OPS");
                else shader.DisableKeyword("VIVID_TSR_WAVE_OPS");
                shader.SetVector("_RenderSize", new Vector4(Size, Size, 1f / Size, 1f / Size));
                shader.SetVector("_OutputSize", new Vector4(Size, Size, 1f / Size, 1f / Size));
                return shader;
            }

            private static Texture2D CreateInput(GraphicsFormat format)
                => new(Size, Size, format, TextureCreationFlags.None) { filterMode = FilterMode.Point, wrapMode = TextureWrapMode.Clamp };

            private static RenderTexture CreateOutput(GraphicsFormat format)
            {
                var texture = new RenderTexture(new RenderTextureDescriptor(Size, Size)
                { graphicsFormat = format, depthStencilFormat = GraphicsFormat.None, enableRandomWrite = true, msaaSamples = 1 });
                if (!texture.Create()) throw new InvalidOperationException("Could not create isolated TSR target.");
                return texture;
            }

            private static void SetConstant(Texture2D texture, int channels, float x, float y = 0)
            {
                var values = new float[PixelCount * channels];
                for (int i = 0; i < PixelCount; i++)
                { values[i * channels] = x; if (channels > 1) values[i * channels + 1] = y; }
                texture.SetPixelData(values, 0); texture.Apply(false, false);
            }

            private static void SetColor(float[] values, int pixel, Color value)
            {
                values[pixel * 4] = value.r; values[pixel * 4 + 1] = value.g;
                values[pixel * 4 + 2] = value.b; values[pixel * 4 + 3] = 1;
            }

            private void SetDepthInputs(Input input)
            {
                var depths = new float[PixelCount];
                var errors = new float[PixelCount];
                for (int pixel = 0; pixel < PixelCount; pixel++)
                { depths[pixel] = 0.5f; errors[pixel] = input.DepthError; }
                if (input.NearbyDepthFeatureOffset != Vector2Int.zero)
                {
                    int featureX = Size / 2 + input.NearbyDepthFeatureOffset.x;
                    int featureY = Size / 2 + input.NearbyDepthFeatureOffset.y;
                    depths[featureY * Size + featureX] = 0.52f;
                    // A .52 sample surrounded by .5 produces a full 3x3 .02 error
                    // patch in TSRDilateVelocity. The receiver stays at depth .5,
                    // with zero center error; patches at the viewport edge truncate.
                    for (int y = Mathf.Max(0, featureY - 1); y <= Mathf.Min(Size - 1, featureY + 1); y++)
                        for (int x = Mathf.Max(0, featureX - 1); x <= Mathf.Min(Size - 1, featureX + 1); x++)
                            errors[y * Size + x] = 0.02f;
                }
                depth.SetPixelData(depths, 0); depth.Apply(false, false);
                depthError.SetPixelData(errors, 0); depthError.Apply(false, false);
            }

            internal Snapshot Run(Input input)
            {
                var colors = new float[PixelCount * 4];
                var histories = new float[PixelCount * 4];
                var resurrections = new float[PixelCount * 4];
                for (int y = 0; y < Size; y++) for (int x = 0; x < Size; x++)
                {
                    int pixel = y * Size + x;
                    Color value = input.ChromaPattern
                        ? ((x + y + input.CheckerPhase) % 2 == 0 ? input.NeighborhoodColorA : input.NeighborhoodColorB)
                        : Gray((x + y + input.CheckerPhase) % 2 == 0 ? input.NeighborhoodHigh : input.NeighborhoodLow);
                    if (pixel == Center) value = input.Current;
                    SetColor(colors, pixel, value);
                    Color historyValue = input.HistoryCheckerboard
                        ? Gray((x + y + input.HistoryCheckerPhase) % 2 == 0 ? input.HistoryCheckerHigh : input.HistoryCheckerLow)
                        : input.History;
                    SetColor(histories, pixel, input.HistoryMatchesInput ? value : historyValue);
                    SetColor(resurrections, pixel, input.Resurrection);
                    resurrections[pixel * 4 + 3] = input.PreviousState;
                    if (input.ForceUpdateState) histories[pixel * 4 + 3] = input.ForcedUpdateState;
                }
                color.SetPixelData(colors, 0); color.Apply(false, false);
                history.SetPixelData(histories, 0); history.Apply(false, false);
                resurrection.SetPixelData(resurrections, 0); resurrection.Apply(false, false);
                SetDepthInputs(input);
                SetConstant(instability, 1, input.LumaInstability);
                SetConstant(motion, 2, input.MotionPixels / Size);
                SetConstant(historyMeta, 2, input.HistorySamples, input.HistoryDepth);
                SetConstant(resurrectionMeta, 2, input.ResurrectionFrames, 0.5f);
                int kernel = reject.FindKernel("CS");
                reject.SetVector("_TSRRejectionParams", new Vector4(0.003f, 16, 0.28f, 0.35f));
                reject.SetTexture(kernel, "_InputColor", color); reject.SetTexture(kernel, "_InputDepth", depth);
                reject.SetTexture(kernel, "_DilatedDepth", depth); reject.SetTexture(kernel, "_DilatedMotion", motion);
                reject.SetTexture(kernel, "_DepthError", depthError); reject.SetTexture(kernel, "_ReprojectionBoundary", zero);
                reject.SetTexture(kernel, "_LumaInstability", instability); reject.SetTexture(kernel, "_ReprojectedHistoryColor", history);
                reject.SetTexture(kernel, "_ReprojectedHistoryMeta", historyMeta);
                reject.SetTexture(kernel, "_ReprojectedResurrectionColor", resurrection);
                reject.SetTexture(kernel, "_AcceptedHistoryColor", acceptedColor); reject.SetTexture(kernel, "_RejectionMask", rejection);
                reject.Dispatch(kernel, 1, 1, 1);
                kernel = update.FindKernel("CS");
                update.SetVector("_TSRParams", new Vector4(input.HistorySamples > 0 ? 1 : 0, 16, 0, 0));
                update.SetTexture(kernel, "_CurrentFrameColor", color); update.SetTexture(kernel, "_DilatedMotion", motion);
                update.SetTexture(kernel, "_DilatedDepth", depth); update.SetTexture(kernel, "_ReprojectionBoundary", zero);
                update.SetTexture(kernel, "_ThinGeometryCoverage", zero); update.SetTexture(kernel, "_LumaInstability", instability);
                update.SetTexture(kernel, "_AcceptedHistoryColor", input.ForceUpdateState ? (Texture)history : acceptedColor);
                update.SetTexture(kernel, "_ReprojectedHistoryMeta", historyMeta);
                update.SetTexture(kernel, "_ReprojectedResurrectionColor", resurrection);
                update.SetTexture(kernel, "_ReprojectedResurrectionMeta", resurrectionMeta); update.SetTexture(kernel, "_RejectionMask", input.ForceUpdateState ? (Texture)zero : rejection);
                update.SetTexture(kernel, "_UpdatedHistoryColor", updatedColor); update.SetTexture(kernel, "_UpdatedHistoryMeta", updatedMeta);
                update.SetTexture(kernel, "_UpdatedResurrectionColor", updatedResurrectionColor);
                update.SetTexture(kernel, "_UpdatedResurrectionMeta", updatedResurrectionMeta);
                update.Dispatch(kernel, 1, 1, 1);
                float[] accepted = Read(acceptedColor), masks = Read(rejection), updated = Read(updatedColor);
                float[] meta = Read(updatedMeta), resurrectionData = Read(updatedResurrectionMeta);
                float[] resurrectionColors = Read(updatedResurrectionColor);
                return new Snapshot
                {
                    CurrentColors = colors, HistoryColors = histories, UpdatedColors = updated, Acceptance = masks, Accepted = input.ForceUpdateState ? 0 : masks[Center],
                    AcceptedAlpha = input.ForceUpdateState ? input.ForcedUpdateState : accepted[Center * 4 + 3], SampleCount = meta[Center * 2],
                    AcceptedHistory = new Color(accepted[Center * 4], accepted[Center * 4 + 1], accepted[Center * 4 + 2], accepted[Center * 4 + 3]),
                    ResurrectionFrames = resurrectionData[Center * 2], PendingState = resurrectionColors[Center * 4 + 3],
                    ResurrectionColor = new Color(resurrectionColors[Center * 4], resurrectionColors[Center * 4 + 1], resurrectionColors[Center * 4 + 2], resurrectionColors[Center * 4 + 3]),
                    Updated = new Color(updated[Center * 4], updated[Center * 4 + 1], updated[Center * 4 + 2], updated[Center * 4 + 3])
                };
            }

            private static float[] Read(RenderTexture texture)
            {
                var request = AsyncGPUReadback.Request(texture, 0); request.WaitForCompletion();
                if (request.hasError) throw new InvalidOperationException("Isolated TSR readback failed.");
                return request.GetData<float>().ToArray();
            }

            public void Dispose()
            {
                foreach (Texture2D texture in new[] { color, history, resurrection, depth, depthError, zero, instability, motion, historyMeta, resurrectionMeta })
                    Object.DestroyImmediate(texture);
                foreach (RenderTexture texture in new[] { acceptedColor, rejection, updatedColor, updatedMeta, updatedResurrectionColor, updatedResurrectionMeta })
                { texture.Release(); Object.DestroyImmediate(texture); }
                Object.DestroyImmediate(reject); Object.DestroyImmediate(update);
            }
        }
    }
