using System;
using System.IO;
using UnityEditor;
using UnityEngine;
using VividRP.Runtime.RenderPass.Core;
namespace VividRP.Editor
{
    internal static class VSMPriorityTiming
    {
        private static VSMBaselineSession session;
        private static double started;
        [MenuItem("Tools/VividRP/Diagnostics/Run Temporary VSM Priority Timing")]
        private static void Run()
        {
            if (Application.dataPath != "E:/VividRP_Reborn/Assets" || !EditorApplication.isPlaying || session != null)
                throw new InvalidOperationException("Use VividRP_Reborn in Play Mode.");
            VirtualShadowMapPrototypeRuntime.ReleaseResources();
            string label = File.ReadAllText(Path.GetFullPath(Path.Combine(Application.dataPath, "../Packages/VividRP/Temp~/vsm-priority-variant.txt"))).Trim();
            var settings = new VSMBaselineCase { name = label, resolution = 4096, firstLevel = 0,
                screenDensity = true, pcf = true, transition = .2f, maxDistance = 150, screenSpaceDenoise = true };
            session = new VSMBaselineSession(Camera.main, new[] { settings }, 5, 5, 120, 4096,
                "Packages/VividRP/Temp~/vsm-priority-timing/" + label, label, false);
            started = EditorApplication.timeSinceStartup;
            EditorApplication.update += Tick;
            AssemblyReloadEvents.beforeAssemblyReload += Finish;
        }
        private static void Tick()
        {
            if (session == null) return;
            if (session.Tick()) { session.Finish("completed"); Finish(); return; }
            if (session.IsFinished || EditorApplication.timeSinceStartup - started > 45) Finish();
        }
        private static void Finish()
        {
            EditorApplication.update -= Tick;
            AssemblyReloadEvents.beforeAssemblyReload -= Finish;
            if (session != null) { session.Dispose(); session = null; }
            if (EditorApplication.isPlaying) EditorApplication.isPlaying = false;
        }
    }
}
