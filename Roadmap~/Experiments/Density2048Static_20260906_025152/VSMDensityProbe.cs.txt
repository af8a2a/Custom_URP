using System;
using System.IO;
using UnityEditor;
using UnityEngine;
using UnityEngine.Experimental.Rendering;
using UnityEngine.Rendering;
using VividRP.Runtime;
using VividRP.Runtime.RenderPass.Core;

namespace VividRP.Editor
{
    internal static class VSMDensityProbe
    {
        private static readonly string[] Names = { "coverage", "density1", "density05", "coverage-repeat" };
        private static ComputeShader shader;
        private static RenderTexture output, data;
        private static GraphicsBuffer projections;
        private static GameObject volumeObject;
        private static VolumeProfile profile;
        private static CascadedShadowSettingsVolume overrides;
        private static string root, directory;
        private static int stage, frames, lastUnityFrame, pending;
        private static double stageStarted;
        private static bool captured, previousBackground;
        private static Vector2 captureJitter;

        [MenuItem("Tools/VividRP/Diagnostics/Capture Temporary VSM Density Comparison")]
        private static void Start()
        {
            if (shader != null) return;
            if (Application.dataPath != "E:/VividRP_Reborn/Assets" || !EditorApplication.isPlaying)
                throw new InvalidOperationException("Density capture requires VividRP_Reborn in Play Mode.");
            root = Path.Combine(Path.GetTempPath(), "vsm-density-probe", DateTime.UtcNow.ToString("HHmmss"));
            Directory.CreateDirectory(root);
            shader = UnityEngine.Object.Instantiate(AssetDatabase.LoadAssetAtPath<ComputeShader>(
                "Packages/com.vivid.render-pipelines/Shaders/Core/Private/CSMShadowResolve.compute"));
            previousBackground = Application.runInBackground;
            Application.runInBackground = true;
            volumeObject = new GameObject("Temporary VSM Density Comparison") { hideFlags = HideFlags.HideAndDontSave };
            var volume = volumeObject.AddComponent<Volume>();
            volume.isGlobal = true; volume.priority = 100000; volume.weight = 1;
            profile = ScriptableObject.CreateInstance<VolumeProfile>();
            overrides = profile.Add<CascadedShadowSettingsVolume>();
            volume.sharedProfile = profile;
            stage = 0; SetStage();
            CSMShadowResolvePass.EditorReceiverCapture += Capture;
            EditorApplication.update += Update;
            AssemblyReloadEvents.beforeAssemblyReload += Cleanup;
        }

        private static void SetStage()
        {
            overrides.virtualShadowMapResolution.Override(2048);
            overrides.virtualShadowMapScreenDensity.Override(stage == 1 || stage == 2);
            overrides.virtualShadowMapTargetTexelPixels.Override(stage == 2 ? 0.5f : 1.0f);
            overrides.virtualShadowMapResolutionLodBias.Override(0);
            directory = Path.Combine(root, Names[stage]); Directory.CreateDirectory(directory);
            frames = 0; lastUnityFrame = -1; captured = false; stageStarted = EditorApplication.timeSinceStartup;
        }

        private static void Capture(ComputePassContext context, Texture depth, Texture normal, Texture shadow)
        {
            var camera = context.Get<VividCameraData>();
            if (captured || camera.camera != Camera.main || lastUnityFrame == Time.frameCount) return;
            lastUnityFrame = Time.frameCount;
            if (++frames < 120) return;
            var cameraSettings = camera.camera.GetComponent<VividAdditionalCameraData>();
            if (stage > 0 && cameraSettings.tsrJitterOffset != captureJitter) return;
            captureJitter = cameraSettings.tsrJitterOffset;
            if (!VirtualShadowMapPrototypeRuntime.HasReceiverDebugSnapshot(
                EntityId.ToULong(camera.camera.GetEntityId()), camera.frameIndex >= 0 ? camera.frameIndex : Time.frameCount)) return;
            captured = true;
            if (output == null)
            {
            var descriptor = new RenderTextureDescriptor(depth.width, depth.height, GraphicsFormat.R32G32B32A32_SFloat, 0)
                { enableRandomWrite = true };
            output = new RenderTexture(descriptor); data = new RenderTexture(descriptor); output.Create(); data.Create();
            }
            var settings = VividVolumeManagerUtility.GetCascadedShadowSettingsVolume();
            var shadowData = context.Get<VividShadowData>();
            var layout = shadowData.clipmaps;
            var records = new VirtualShadowMapProjection[layout.Count];
            if (projections == null) projections = new GraphicsBuffer(GraphicsBuffer.Target.Structured, layout.Count, 160);
            var cmd = context.cmd.m_WrappedCommandBuffer;
            int kernel = shader.FindKernel("VSMReceiverDebug");
            cmd.SetComputeTextureParam(shader, kernel, "_DepthTexture", depth);
            cmd.SetComputeTextureParam(shader, kernel, "_GBuffer1", normal);
            cmd.SetComputeTextureParam(shader, kernel, "_VSMReceiverDebugShadow", shadow);
            cmd.SetComputeTextureParam(shader, kernel, "_VSMReceiverDebugOutput", output);
            cmd.SetComputeTextureParam(shader, kernel, "_VSMReceiverDebugData", data);
            cmd.SetComputeTextureParam(shader, kernel, "_VSMPrototypeStaticPhysicalPage", VirtualShadowMapPrototypeRuntime.StaticPhysicalPage);
            cmd.SetComputeTextureParam(shader, kernel, "_VSMPrototypeDynamicPhysicalPage", VirtualShadowMapPrototypeRuntime.DynamicPhysicalPage);
            cmd.SetComputeBufferParam(shader, kernel, "_VSMPrototypePageTable", VirtualShadowMapPrototypeRuntime.PageTable);
            cmd.SetComputeBufferParam(shader, kernel, "_VSMPrototypePageMetadata", VirtualShadowMapPrototypeRuntime.PageMetadata);
            cmd.SetComputeBufferParam(shader, kernel, "_VSMProjections", projections);
            cmd.SetComputeIntParam(shader, "_VSMProjectionCount", layout.Count);
            var vp = camera.GetGPUViewProjectionMatrix(true);
            cmd.SetComputeMatrixParam(shader, "_VSMReceiverViewProjection", vp);
            cmd.SetComputeMatrixParam(shader, "_CSMInvViewProjMatrix", vp.inverse);
            cmd.SetComputeVectorParam(shader, "_VSMReceiverQuality", VirtualShadowMapReceiverQuality.BuildParameters(settings));
            cmd.SetComputeIntParam(shader, "_CSMOutputWidth", depth.width); cmd.SetComputeIntParam(shader, "_CSMOutputHeight", depth.height);
            cmd.SetComputeIntParam(shader, "_VSMPrototypeEnabled", 1);
            cmd.SetComputeIntParam(shader, "_VSMPrototypeVirtualResolution", VirtualShadowMapPrototypeRuntime.VirtualResolution);
            cmd.SetComputeIntParam(shader, "_VSMPrototypePageSize", VirtualShadowMapPrototypeRuntime.PageSize);
            cmd.SetComputeIntParam(shader, "_VSMPrototypePagesPerAxis", VirtualShadowMapPrototypeRuntime.PagesPerAxis);
            cmd.SetComputeIntParam(shader, "_VSMPrototypePhysicalPagesPerRow", VirtualShadowMapPrototypeRuntime.PhysicalPagesPerRow);
            for (int i = 0; i < layout.Count; i++)
            {
                var center = layout.CameraPosition;
                records[i] = new VirtualShadowMapProjection {
                    WorldToClip = GL.GetGPUProjectionMatrix(layout.Projections[i], true) * layout.Views[i],
                    WorldToShadow = VividShadowData.BuildWorldToShadowMatrix(layout.Projections[i], layout.Views[i]),
                    SelectionSphere = new Vector4(center.x, center.y, center.z, -layout.Radii[i]),
                    Parameters = new Vector4(2 * layout.Radii[i] / layout.Resolution, layout.NormalBias, layout.BlendBorder, layout.MaxDistance)
                };
            }
            cmd.SetBufferData(projections, records);
            cmd.SetComputeVectorParam(shader, "_VSMReceiverParameters", new Vector4(
                settings.virtualShadowMapPCF.value ? 1 : 0, shadowData.depthBias, shadowData.slopeScaleDepthBias, 0));
            foreach (int mode in new[] { 0, 3, 5, 6 })
            {
                string path = Path.Combine(directory, "mode" + mode + ".bin");
                cmd.SetComputeIntParam(shader, "_VSMReceiverDebugMode", mode);
                cmd.DispatchCompute(shader, kernel, (depth.width + 7) / 8, (depth.height + 7) / 8, 1);
                pending++;
                cmd.RequestAsyncReadback(data, 0, request => {
                    try { if (!request.hasError) File.WriteAllBytes(path, request.GetData<byte>().ToArray()); }
                    finally { pending--; }
                });
            }
            SaveBuffer(cmd, VirtualShadowMapPrototypeRuntime.AllocatorCounters, "counters.bin");
            SaveBuffer(cmd, VirtualShadowMapPrototypeRuntime.PageTable, "table.bin");
            SaveBuffer(cmd, VirtualShadowMapPrototypeRuntime.PageMetadata, "metadata.bin");
            File.WriteAllText(Path.Combine(directory, "frame.txt"), "project=" + Application.dataPath
                + "\nposition=" + camera.camera.transform.position.ToString("R")
                + "\nrotation=" + camera.camera.transform.eulerAngles.ToString("R")
                + "\nsize=" + depth.width + "x" + depth.height + "\nlevels=" + layout.Count
                + "\nresolution=" + layout.Resolution + "\nfirstLevel=" + layout.FirstLevel
                + "\ncameraFrame=" + camera.frameIndex + "\njitter=" + captureJitter.ToString("R")
                + "\neffectiveAA=" + context.Get<VividAntialiasingData>().effectiveMode
                + "\ngpuVP=" + vp.ToString("R")
                + "\nframe=" + Time.frameCount + "\ncapacity=" + VirtualShadowMapPrototypeRuntime.PhysicalPageCapacity
                + "\nstate=" + VirtualShadowMapPrototypeRuntime.FrameState
                + "\nfallback=" + VirtualShadowMapPrototypeRuntime.LastFallbackReason
                + "\nsettings=" + EditorJsonUtility.ToJson(settings));
            ScreenCapture.CaptureScreenshot(Path.Combine(directory, "game.png"));
        }

        private static void SaveBuffer(CommandBuffer cmd, GraphicsBuffer buffer, string file)
        {
            string path = Path.Combine(directory, file); pending++;
            cmd.RequestAsyncReadback(buffer, request => {
                try { if (!request.hasError) File.WriteAllBytes(path, request.GetData<byte>().ToArray()); }
                finally { pending--; }
            });
        }

        private static void Update()
        {
            if (EditorApplication.timeSinceStartup - stageStarted > 90)
            {
                File.WriteAllText(Path.Combine(root, "status.txt"), "timeout stage=" + stage + " frames=" + frames);
                Cleanup(); EditorApplication.isPlaying = false; return;
            }
            if (!captured || pending > 0 || !File.Exists(Path.Combine(directory, "game.png"))) return;
            File.WriteAllText(Path.Combine(directory, "status.txt"), "captured=True frames=" + frames + " pending=" + pending);
            if (++stage < Names.Length) SetStage();
            else
            {
                File.WriteAllText(Path.Combine(root, "status.txt"), "complete");
                Cleanup(); EditorApplication.isPlaying = false;
            }
        }

        private static void Cleanup()
        {
            CSMShadowResolvePass.EditorReceiverCapture -= Capture;
            EditorApplication.update -= Update; AssemblyReloadEvents.beforeAssemblyReload -= Cleanup;
            AsyncGPUReadback.WaitAllRequests();
            if (output != null) { output.Release(); data.Release(); UnityEngine.Object.DestroyImmediate(output); UnityEngine.Object.DestroyImmediate(data); }
            projections?.Dispose(); projections = null;
            if (volumeObject != null) UnityEngine.Object.DestroyImmediate(volumeObject);
            if (profile != null) UnityEngine.Object.DestroyImmediate(profile);
            Application.runInBackground = previousBackground;
            UnityEngine.Object.DestroyImmediate(shader); shader = null; output = null; data = null;
        }
    }
}
