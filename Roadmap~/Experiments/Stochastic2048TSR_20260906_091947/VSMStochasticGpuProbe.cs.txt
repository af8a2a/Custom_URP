using System;
using System.IO;
using Unity.Mathematics;
using UnityEditor;
using UnityEngine;
using UnityEngine.Experimental.Rendering;
using UnityEngine.Rendering;
using VividRP.Runtime;
using VividRP.Runtime.RenderPass.Core;
using Object = UnityEngine.Object;

namespace VividRP.Editor
{
    // Temporary isolated GPU diagnostic. It owns every resource it touches and
    // dispatches the shipping sampler through its small diagnostic kernels.
    public static class VSMStochasticGpuProbe
    {
        [Serializable]
        private sealed class Report
        {
            public bool success;
            public string error, device, output;
            public int samples, changedFrames, fallbackFrames, slopeFrames, roundingBoundaryCases;
            public float meanRadiusSquared, meanOffsetLength, pixelDifferenceSquared;
            public double diskCoverage, diskMean, individualMse, mean64FrameMse;
            public float minimumCorrectedShadow = 1, maximumBlockerShadow;
            public float minimumDefaultBiasCoplanar = 1, maximumDefaultBiasBlocker;
        }

        [MenuItem("Tools/VividRP/Diagnostics/Run Temporary VSM Stochastic GPU Probe")]
        public static void Run()
        {
            if (!Application.dataPath.Replace('\\', '/').Equals("E:/VividRP_Reborn/Assets", StringComparison.OrdinalIgnoreCase))
                throw new InvalidOperationException("VSM stochastic probe is scoped to VividRP_Reborn.");
            var report = new Report { device = SystemInfo.graphicsDeviceType.ToString() };
            report.output = Path.Combine(Path.GetTempPath(), "vsm-stochastic-gpu-" + DateTime.UtcNow.ToString("yyyyMMdd-HHmmss") + ".json");
            try
            {
                using var gpu = new Probe();
                InspectSamples(gpu, report);
                InspectDiskMean(gpu, report);
                InspectMissingPages(gpu, report);
                InspectPlanes(gpu, report);
                report.success = true;
            }
            catch (Exception exception) { report.error = exception.ToString(); }
            File.WriteAllText(report.output, JsonUtility.ToJson(report, true));
            if (report.success) Debug.Log("VSM stochastic GPU diagnostic: " + report.output);
            else Debug.LogError("VSM stochastic GPU diagnostic failed: " + report.output + "\n" + report.error);
        }

        private static void Check(bool valid, string description)
        {
            if (!valid) throw new InvalidOperationException(description);
        }

        private static void InspectSamples(Probe gpu, Report report)
        {
            var boundaries = gpu.Run("InspectVSMStochasticTexelOffset", new[]
            {
                new float4(0.49999994f, 0.49999994f, 1, 0), new float4(0.49999994f, 0.49999994f, 0, 1),
                new float4(-0.5f, -0.5f, -1, 0), new float4(-0.5f, -0.5f, 0, -1),
                new float4(-0.2f, 0.2f, 0.6f, -0.8f),
            });
            var expected = new[] { new float2(1, 0), new float2(0, 1), new float2(-1, 0), new float2(0, -1), new float2(0, -1) };
            for (int i = 0; i < boundaries.Length; i++)
                Check(math.all(boundaries[i] == expected[i]), "Floating-point rounding escaped the one-texel feedback halo.");
            report.roundingBoundaryCases = boundaries.Length;
            var inputs = new float4[128 * 9];
            for (int frame = 0; frame < 128; frame++) for (int sample = 0; sample < 9; sample++)
                inputs[frame * 9 + sample] = new float4(17, 29, sample, frame);
            var samples = gpu.Run("InspectVSMStochasticSample", inputs);
            var repeated = gpu.Run("InspectVSMStochasticSample", inputs);
            float2 mean = 0;
            float radiusSum = 0;
            for (int i = 0; i < samples.Length; i++)
            {
                int sample = i % 9;
                Check(math.all(samples[i] == repeated[i]), "Pixel/frame repeat changed sample.");
                float radiusSquared = math.lengthsq(samples[i]);
                float angle = math.atan2(samples[i].y, samples[i].x);
                if (angle < 0) angle += math.PI * 2;
                Check(radiusSquared >= sample / 3 / 3f - 1e-6f && radiusSquared <= (sample / 3 + 1) / 3f + 1e-6f,
                    "Sample escaped its equal-area radial stratum.");
                Check(angle >= sample % 3 * math.PI * 2 / 3 - 1e-6f && angle <= (sample % 3 + 1) * math.PI * 2 / 3 + 1e-6f,
                    "Sample escaped its angular stratum.");
                mean += samples[i]; radiusSum += radiusSquared;
                if (i >= 9 && math.lengthsq(samples[i] - samples[i - 9]) > 1e-6f) report.changedFrames++;
            }
            inputs[0].x += 1;
            report.pixelDifferenceSquared = math.lengthsq(gpu.Run("InspectVSMStochasticSample", new[] { inputs[0] })[0] - samples[0]);
            report.samples = samples.Length;
            report.meanRadiusSquared = radiusSum / samples.Length;
            report.meanOffsetLength = math.length(mean / samples.Length);
            Check(report.changedFrames > 0 && report.pixelDifferenceSquared > 1e-6f, "Random phase did not change.");
            Check(report.meanOffsetLength < 0.04f && Mathf.Abs(report.meanRadiusSquared - 0.5f) < 0.02f,
                "Disk samples have unexpected spatial bias.");
        }

        private static void InspectDiskMean(Probe gpu, Report report)
        {
            gpu.Clear();
            for (int page = 0; page < 4; page++) gpu.Map(page, page % 2 == 0 ? 0.8f : 0.2f);
            gpu.Upload();
            var input = new[] { new float4(3.75f / 8, 4.2f / 8, 0.5f, 0) };
            report.diskCoverage = (Math.Acos(0.25) - 0.25 * Math.Sqrt(1 - 0.25 * 0.25)) / Math.PI;
            double sum = 0, block = 0, individual = 0, accumulated = 0;
            for (int frame = 0; frame < 512; frame++)
            {
                gpu.Shader.SetInt("_CSMFrameIndex", frame);
                float2 result = gpu.Run("FilterFootprints", input)[0];
                Check(result.x == 1 && Mathf.Abs(result.y * 9 - Mathf.Round(result.y * 9)) < 1e-5f,
                    "Result is not a complete nine-comparison mean.");
                sum += result.y; block += result.y;
                individual += (result.y - report.diskCoverage) * (result.y - report.diskCoverage);
                if ((frame + 1) % 64 == 0)
                {
                    double error = block / 64 - report.diskCoverage;
                    accumulated += error * error; block = 0;
                }
            }
            report.diskMean = sum / 512; report.individualMse = individual / 512; report.mean64FrameMse = accumulated / 8;
            Check(Math.Abs(report.diskMean - report.diskCoverage) < 0.025, "Temporal mean missed analytic disk coverage.");
            Check(report.mean64FrameMse < report.individualMse * 0.15, "Temporal accumulation failed to reduce sampling variance.");
        }

        private static void InspectMissingPages(Probe gpu, Report report)
        {
            var input = new[] { new float4(-0.8125f, -0.8125f, 0, 0) };
            for (int dirty = 0; dirty < 2; dirty++)
            {
                gpu.Clear();
                for (int page = 0; page < 3; page++) gpu.Map(page, 0.8f);
                if (dirty != 0) { gpu.Map(3, 0.8f); gpu.Metadata[3].x |= 4; }
                for (int page = 4; page < 8; page++) gpu.Map(page, 0);
                gpu.Upload();
                for (int frame = 0; frame < 64; frame++)
                {
                    gpu.Shader.SetInt("_CSMFrameIndex", frame);
                    Check(gpu.Run("ResolveReceivers", input)[0].x == 1, "A random phase used an incomplete fine footprint.");
                    report.fallbackFrames++;
                }
                gpu.Shader.SetVector("_VSMReceiverParameters", new Vector4(0, 0, 0, 1));
                Check(gpu.Run("ResolveReceivers", input)[0].x == 0, "Stochastic toggle changed hard-shadow behavior.");
            }
        }

        private static void InspectPlanes(Probe gpu, Report report)
        {
            gpu.Clear();
            for (int page = 0; page < 4; page++) gpu.Map(page, 0);
            for (int y = 0; y < 8; y++) for (int x = 0; x < 8; x++)
                gpu.SetDepth(x, y, 0.5f + 0.05f * (x - 2.75f) + 0.03f * (y - 2.75f));
            gpu.Upload();
            var input = new[] { new float4(3.25f / 8, 3.25f / 8, 0.5f, 0) };
            var correction = new[] { new float4(0.05f, 0.03f, 1e-5f, 0) };
            for (int blocked = 0; blocked < 2; blocked++)
            {
                for (int frame = 0; frame < 64; frame++)
                {
                    gpu.Shader.SetInt("_CSMFrameIndex", frame);
                    float2 value = gpu.Run("FilterFootprints", input, correction)[0];
                    Check(value.x == 1, "Plane diagnostic lost a page.");
                    if (blocked == 0) report.minimumCorrectedShadow = Mathf.Min(report.minimumCorrectedShadow, value.y);
                    else report.maximumBlockerShadow = Mathf.Max(report.maximumBlockerShadow, value.y);
                    report.slopeFrames++;
                }
                for (int y = 0; y < 8; y++) for (int x = 0; x < 8; x++)
                    gpu.SetDepth(x, y, 0.54f + 0.05f * (x - 2.75f) + 0.03f * (y - 2.75f));
                gpu.Upload();
            }
            Check(report.minimumCorrectedShadow > 0.99999f && report.maximumBlockerShadow == 0,
                "Actual texel-center plane correction failed coplanar/blocker separation.");

            const float texel = 0.015625f;
            gpu.Clear();
            gpu.Shader.SetInt("_VSMProjectionCount", 1);
            gpu.Shader.SetVector("_VSMReceiverParameters", new Vector4(1, VividAdditionalLightData.DefaultShadowDepthBias,
                VividAdditionalLightData.DefaultShadowSlopeBias, 1));
            gpu.Projections[0].WorldToShadow.m00 = gpu.Projections[0].WorldToShadow.m11 = 1 / (8 * texel);
            gpu.Projections[0].WorldToShadow.m22 = 0.1f;
            gpu.Projections[0].Parameters = new Vector4(texel, VividAdditionalLightData.DefaultShadowNormalBias, 0, 100);
            for (int page = 0; page < 4; page++) gpu.Map(page, 0);
            var receiver = new[] { float4.zero };
            var normal = new[] { new float4(-1, 0, 1, 0) };
            for (int blocked = 0; blocked < 2; blocked++)
            {
                for (int y = 0; y < 8; y++) for (int x = 0; x < 8; x++)
                    gpu.SetDepth(x, y, 0.5f + 0.1f * (x - 3.5f + blocked * 3) * texel);
                gpu.Upload();
                for (int frame = 0; frame < 64; frame++)
                {
                    gpu.Shader.SetInt("_CSMFrameIndex", frame);
                    float value = gpu.Run("ResolveReceivers", receiver, normal)[0].x;
                    if (blocked == 0) report.minimumDefaultBiasCoplanar = Mathf.Min(report.minimumDefaultBiasCoplanar, value);
                    else report.maximumDefaultBiasBlocker = Mathf.Max(report.maximumDefaultBiasBlocker, value);
                    report.slopeFrames++;
                }
            }
            Check(report.minimumDefaultBiasCoplanar > 0.99999f && report.maximumDefaultBiasBlocker == 0,
                "Default residual slope bias regressed near-blocker shadows.");
        }

        private sealed class Probe : IDisposable
        {
            internal readonly ComputeShader Shader;
            internal readonly uint4[] Metadata = new uint4[12];
            internal readonly VirtualShadowMapProjection[] Projections = new VirtualShadowMapProjection[3];
            private readonly uint[] table = new uint[12], depths = new uint[256];
            private readonly GraphicsBuffer tableBuffer = new(GraphicsBuffer.Target.Structured, 12, 4);
            private readonly GraphicsBuffer metadataBuffer = new(GraphicsBuffer.Target.Structured, 12, 16);
            private readonly GraphicsBuffer projectionsBuffer = new(GraphicsBuffer.Target.Structured, 3, 160);
            private readonly GraphicsBuffer upload = new(GraphicsBuffer.Target.Structured, 256, 4);
            private readonly RenderTexture staticPool, dynamicPool;
            private static readonly int[] Slots = { 9, 2, 14, 6, 0, 1, 3, 4, 5, 7, 8, 10 };

            internal Probe()
            {
                Shader = Object.Instantiate(AssetDatabase.LoadAssetAtPath<ComputeShader>(
                    "Packages/com.vivid.render-pipelines/Tests/Editor/RenderPass/Shadows/VirtualShadowMapSamplingTests.compute"));
                var descriptor = new RenderTextureDescriptor(16, 16)
                {
                    graphicsFormat = GraphicsFormat.R32_UInt, depthStencilFormat = GraphicsFormat.None,
                    enableRandomWrite = true, msaaSamples = 1
                };
                staticPool = new RenderTexture(descriptor); dynamicPool = new RenderTexture(descriptor);
                Check(staticPool.Create() && dynamicPool.Create(), "Could not create isolated diagnostic pools.");
                Shader.SetInt("_VSMPrototypeEnabled", 1); Shader.SetInt("_VSMPrototypeRequestEnabled", 1);
                Shader.SetInt("_VSMProjectionCount", 3); Shader.SetInt("_VSMPrototypeVirtualResolution", 8);
                Shader.SetInt("_VSMPrototypePageSize", 4); Shader.SetInt("_VSMPrototypePagesPerAxis", 2);
                Shader.SetInt("_VSMPrototypePhysicalPagesPerRow", 4); Shader.SetInt("_VSMPrototypePageTableEntryCount", 12);
                Shader.SetInt("_VSMPrototypePhysicalPageCapacity", 16); Shader.SetInt("_VSMPrototypeFeedbackFrameIndex", 7);
                Shader.SetInt("_CSMOutputWidth", 8); Shader.SetInt("_CSMOutputHeight", 8);
                Shader.SetInts("_SamplingPixel", 17, 29);
                Shader.SetMatrix("_VSMReceiverViewProjection", Matrix4x4.identity);
                Shader.SetVector("_VSMReceiverQuality", Vector4.zero);
                Clear();
            }

            internal void Clear()
            {
                Array.Clear(table, 0, table.Length); Array.Clear(Metadata, 0, Metadata.Length); Array.Clear(depths, 0, depths.Length);
                Shader.SetVector("_VSMReceiverParameters", new Vector4(1, 0, 0, 1));
                for (int i = 0; i < 3; i++)
                {
                    var matrix = Matrix4x4.identity;
                    matrix.m00 = matrix.m11 = 0.1f / (1 << i); matrix.m03 = matrix.m13 = matrix.m23 = 0.5f;
                    Projections[i] = new VirtualShadowMapProjection
                    {
                        WorldToShadow = matrix, SelectionSphere = new Vector4(0, 0, 0, -5 * (1 << i)),
                        Parameters = new Vector4(1 << i, 0, 0.1f, 100)
                    };
                }
            }

            internal void Map(int page, float depth)
            {
                int slot = Slots[page]; table[page] = (uint)slot + 1; Metadata[page] = new uint4(10, (uint)slot + 1, 7, 8);
                for (int y = 0; y < 4; y++) for (int x = 0; x < 4; x++)
                    depths[(slot / 4 * 4 + y) * 16 + slot % 4 * 4 + x] = math.asuint(depth);
            }

            internal void SetDepth(int x, int y, float depth)
            {
                int slot = Slots[y / 4 * 2 + x / 4];
                depths[(slot / 4 * 4 + y % 4) * 16 + slot % 4 * 4 + x % 4] = math.asuint(depth);
            }

            internal void Upload()
            {
                tableBuffer.SetData(table); metadataBuffer.SetData(Metadata); projectionsBuffer.SetData(Projections);
                int kernel = Shader.FindKernel("UploadTestPools");
                using var empty = new GraphicsBuffer(GraphicsBuffer.Target.Structured, 256, 4);
                empty.SetData(new uint[256]); upload.SetData(depths);
                Shader.SetBuffer(kernel, "_TestStaticData", upload); Shader.SetBuffer(kernel, "_TestDynamicData", empty);
                Shader.SetTexture(kernel, "_TestStaticPool", staticPool); Shader.SetTexture(kernel, "_TestDynamicPool", dynamicPool);
                Shader.Dispatch(kernel, 2, 2, 1);
            }

            internal float2[] Run(string name, float4[] inputs, float4[] normals = null)
            {
                int kernel = Shader.FindKernel(name);
                using var input = new GraphicsBuffer(GraphicsBuffer.Target.Structured, inputs.Length, 16);
                using var normal = new GraphicsBuffer(GraphicsBuffer.Target.Structured, inputs.Length, 16);
                using var output = new GraphicsBuffer(GraphicsBuffer.Target.Structured, inputs.Length, 8);
                input.SetData(inputs); normal.SetData(normals ?? new float4[inputs.Length]);
                Shader.SetInt("_SamplingCount", inputs.Length);
                Shader.SetBuffer(kernel, "_SamplingInputs", input); Shader.SetBuffer(kernel, "_SamplingResults", output);
                if (name != "InspectVSMStochasticSample" && name != "InspectVSMStochasticTexelOffset")
                {
                    Shader.SetBuffer(kernel, "_SamplingNormals", normal);
                    Shader.SetBuffer(kernel, "_VSMPrototypePageTable", tableBuffer);
                    Shader.SetBuffer(kernel, "_VSMPrototypePageMetadata", metadataBuffer);
                    Shader.SetTexture(kernel, "_VSMPrototypeStaticPhysicalPage", staticPool);
                    Shader.SetTexture(kernel, "_VSMPrototypeDynamicPhysicalPage", dynamicPool);
                    if (name == "ResolveReceivers") Shader.SetBuffer(kernel, "_VSMProjections", projectionsBuffer);
                }
                Shader.Dispatch(kernel, (inputs.Length + 63) / 64, 1, 1);
                var result = new float2[inputs.Length]; output.GetData(result); return result;
            }

            public void Dispose()
            {
                tableBuffer.Dispose(); metadataBuffer.Dispose(); projectionsBuffer.Dispose(); upload.Dispose();
                staticPool.Release(); dynamicPool.Release(); Object.DestroyImmediate(staticPool); Object.DestroyImmediate(dynamicPool);
                Object.DestroyImmediate(Shader);
            }
        }
    }
}
